+"""Inactive CI fixture: shared plan, two serial shards, and the strict gate."""
+
+import argparse
+import json
+import subprocess
+import sys
+from pathlib import Path
+
+
+def normalized(path):
+    rows = []
+    text = path.read_text(encoding="utf-8")
+    if not text.endswith("\n"):
+        raise ValueError(f"missing or truncated self-check evidence: {path}")
+    for line in text.splitlines():
+        row = json.loads(line)
+        del row["wall_time_seconds"]
+        # GitHub's checkout roots need not be identical across jobs.
+        filename = Path(row["file_path"])
+        if not filename.is_absolute() or filename.parent.parts[-2:] != ("src", "pytest_leela"):
+            raise ValueError(f"unexpected self-check target path: {filename}")
+        row["file_path"] = filename.name
+        rows.append(row)
+    return sorted(rows, key=lambda row: row["mutant_id"])
+
+
+def main():
+    parser = argparse.ArgumentParser(description=__doc__)
+    commands = parser.add_subparsers(dest="stage", required=True)
+    plan = commands.add_parser("plan")
+    plan.add_argument("plan", type=Path)
+    serial = commands.add_parser("serial")
+    serial.add_argument("jsonl", type=Path)
+    shard = commands.add_parser("shard")
+    shard.add_argument("plan", type=Path)
+    shard.add_argument("index", choices=("0", "1"))
+    shard.add_argument("jsonl", type=Path)
+    verify = commands.add_parser("verify")
+    verify.add_argument("plan", type=Path)
+    verify.add_argument("shard0", type=Path)
+    verify.add_argument("shard1", type=Path)
+    parity = commands.add_parser("parity")
+    parity.add_argument("plan", type=Path)
+    parity.add_argument("serial", type=Path)
+    parity.add_argument("shard0", type=Path)
+    parity.add_argument("shard1", type=Path)
+    args = parser.parse_args()
+
+    if args.stage in {"verify", "parity"}:
+        # Keep the usual strict self-mutation gate. Do not mask verifier exit 1
+        # (complete but failed mutation policy) or exit 2 (invalid evidence).
+        result = subprocess.run(
+            [sys.executable, "-m", "pytest_leela.shard", str(args.plan),
+             str(args.shard0), str(args.shard1)], check=False,
+        )
+        if args.stage == "verify" or result.returncode not in (0, 1):
+            return result.returncode
+        reference = normalized(args.serial)
+        union = sorted(normalized(args.shard0) + normalized(args.shard1),
+                       key=lambda row: row["mutant_id"])
+        if reference != union:
+            parser.error("serial and shard identities/verdicts differ")
+        print(f"Exact full-suite identity and verdict equality: {len(reference)} occurrences")
+        return result.returncode
+
+    # Same target universe as the existing Python package self-check.
+    targets = sorted(p for p in Path("src/pytest_leela").glob("*.py")
+                     if p.name != "__init__.py")
+    if not targets:
+        parser.error("no self-mutation targets; run from the checkout root")
+    argv = [sys.executable, "-m", "pytest", "--leela"]
+    for target in targets:
+        argv.extend(("--target", str(target)))
+    if args.stage == "plan":
+        argv.extend(("--leela-write-plan", str(args.plan), "--leela-shard-count", "2"))
+    elif args.stage == "serial":
+        argv.extend(("--leela-jsonl", str(args.jsonl)))
+    else:
+        argv.extend(("--leela-plan", str(args.plan), "--leela-shard", f"{args.index}/2",
+                     "--leela-jsonl", str(args.jsonl)))
+    # Keep the raw exit status. Workflows must upload completed artifacts even
+    # on exit 1 and always run the verifier, not declare success from a worker.
+    return subprocess.run(argv, check=False).returncode
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())
