"""Enumerate native mutation points and default/all replacements."""
import dataclasses
import hashlib
import json
import sys
from pathlib import Path

from pytest_leela.ast_analysis import find_mutation_points
from pytest_leela.git_diff import changed_lines
from pytest_leela.operators import DEFAULT_OPERATORS, build_allowed_keys, count_pruned, mutations_for
from pytest_leela.type_extractor import enrich_mutation_points

FILES = [
    'src/pytest_leela/daemon.py',
    'src/pytest_leela/index.py',
    'tests/leela/describe_daemon.py',
    'tests/leela/describe_index.py',
]


def build_report(changes, files):
    report = {}
    for name in files:
        path = Path(name).resolve()
        source = path.read_text()
        points, _stats = enrich_mutation_points(
            source,
            find_mutation_points(source, str(path), name.removesuffix('.py').replace('/', '.')),
        )
        points = [point for point in points if point.lineno in changes[str(path)]]
        default = build_allowed_keys(DEFAULT_OPERATORS)
        report[name] = {
            'sha256': hashlib.sha256(source.encode()).hexdigest(),
            'changed_lines': sorted(changes[str(path)]),
            'default_pruned': count_pruned(points, True, default),
            'all_pruned': count_pruned(points, True),
            'points': [
                {
                    'point': dataclasses.asdict(point),
                    'default_mutations': mutations_for(point, True, default),
                    'all_mutations': mutations_for(point, True),
                }
                for point in points
            ],
        }
    return report


def main(input_stream=None):
    if input_stream is None:
        files = FILES
    else:
        files = json.load(input_stream)
    print(json.dumps(build_report(changed_lines('HEAD'), files), indent=2))


if __name__ == '__main__':
    main(sys.stdin)
